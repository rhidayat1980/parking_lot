import domain/floor

// MARK: Errors

pub type EntryParkingError {
  NoAvailableParkingSlot
  AllocationFailed(parking_allocation.ParkingAllocationError)
}

pub fn error_message(error: EntryParkingError) -> String {
  case error {
    NoAvailableParkingSlot -> "No available parking slot for this vehicle"

    AllocationFailed(allocation_error) ->
      parking_allocation.error_message(allocation_error)
  }
}

// MARK: Entry Parking Request

pub type EntryParkingRequest {
  EntryParkingRequest(
    session_id: parking_session.ParkingSessionId,
    ticket_id: parking_ticket.ParkingTicketId,
    vehicle: vehicle.Vehicle,
    entry_time: Int,
  )
}

// MARK: Entry Parking Result

pub type EntryParkingResult {
  EntryParkingResult(
    floor_id: floor.ParkingFloorId,
    slot: parking_slot.ParkingSlot,
    session: parking_session.ParkingSession,
    ticket: parking_ticket.ParkingTicket,
  )
}

// MARK: Entry Parking

pub fn enter_parking(
  floors: List(floor.ParkingFloor),
  existing_sessions: List(parking_session.ParkingSession),
  request: EntryParkingRequest,
) -> Result(EntryParkingResult, EntryParkingError) {
  let EntryParkingRequest(
    session_id: session_id,
    ticket_id: ticket_id,
    vehicle: candidate_vehicle,
    entry_time: entry_time,
  ) = request

  use #(floor_id, slot, session) <- result.try(find_available_slot(
    floors,
    existing_sessions,
    session_id,
    candidate_vehicle,
    entry_time,
  ))

  let ticket =
    parking_ticket.new_parking_ticket(ticket_id, session_id, entry_time)

  Ok(EntryParkingResult(
    floor_id: floor_id,
    slot: slot,
    session: session,
    ticket: ticket,
  ))
}

// MARK: Slot Search

fn find_available_slot(
  floors: List(floor.ParkingFloor),
  existing_sessions: List(parking_session.ParkingSession),
  session_id: parking_session.ParkingSessionId,
  candidate_vehicle: vehicle.Vehicle,
  entry_time: Int,
) -> Result(
  #(
    floor.ParkingFloorId,
    parking_slot.ParkingSlot,
    parking_session.ParkingSession,
  ),
  EntryParkingError,
) {
  case floors {
    [] -> Error(NoAvailableParkingSlot)

    [current_floor, ..remaining_floors] -> {
      let floor_id = floor.get_parking_floor_id(current_floor)
      let slots = floor.get_parking_floor_slots(current_floor)

      case
        find_available_slot_in_floor(
          slots,
          floor_id,
          existing_sessions,
          session_id,
          candidate_vehicle,
          entry_time,
        )
      {
        Ok(allocation) -> Ok(allocation)

        Error(NoAvailableParkingSlot) ->
          find_available_slot(
            remaining_floors,
            existing_sessions,
            session_id,
            candidate_vehicle,
            entry_time,
          )

        Error(error) -> Error(error)
      }
    }
  }
}

// MARK: Floor Slot Search

fn find_available_slot_in_floor(
  slots: List(parking_slot.ParkingSlot),
  floor_id: floor.ParkingFloorId,
  existing_sessions: List(parking_session.ParkingSession),
  session_id: parking_session.ParkingSessionId,
  candidate_vehicle: vehicle.Vehicle,
  entry_time: Int,
) -> Result(
  #(
    floor.ParkingFloorId,
    parking_slot.ParkingSlot,
    parking_session.ParkingSession,
  ),
  EntryParkingError,
) {
  case slots {
    [] -> Error(NoAvailableParkingSlot)

    [current_slot, ..remaining_slots] ->
      try_allocate_slot(
        current_slot,
        remaining_slots,
        floor_id,
        existing_sessions,
        session_id,
        candidate_vehicle,
        entry_time,
      )
  }
}

// MARK: Slot Allocation Attempt

fn try_allocate_slot(
  slot: parking_slot.ParkingSlot,
  remaining_slots: List(parking_slot.ParkingSlot),
  floor_id: floor.ParkingFloorId,
  existing_sessions: List(parking_session.ParkingSession),
  session_id: parking_session.ParkingSessionId,
  candidate_vehicle: vehicle.Vehicle,
  entry_time: Int,
) -> Result(
  #(
    floor.ParkingFloorId,
    parking_slot.ParkingSlot,
    parking_session.ParkingSession,
  ),
  EntryParkingError,
) {
  let request =
    parking_allocation.ParkingAllocationRequest(
      session_id: session_id,
      vehicle: candidate_vehicle,
      slot: slot,
      entry_time: entry_time,
    )

  case parking_allocation.allocate_parking_session(existing_sessions, request) {
    Ok(session) -> Ok(#(floor_id, slot, session))

    Error(parking_allocation.VehicleTypeNotSupported) ->
      find_available_slot_in_floor(
        remaining_slots,
        floor_id,
        existing_sessions,
        session_id,
        candidate_vehicle,
        entry_time,
      )

    Error(parking_allocation.ParkingSlotOccupied) ->
      find_available_slot_in_floor(
        remaining_slots,
        floor_id,
        existing_sessions,
        session_id,
        candidate_vehicle,
        entry_time,
      )

    Error(error) -> Error(AllocationFailed(error))
  }
}

      rollback_error: rollback_error,
    )) ->
      Error(TransactionRollbackFailed(
        operation_error: operation_error,
        rollback_error: rollback_error,
      ))
  }
}

import domain/parking_fee
import domain/parking_session
import domain/payment
import gleam/option
import gleam/result

// MARK: Errors

pub type ExitParkingError {
  SessionNotFound
  SessionClosingFailed(parking_session.ParkingSessionError)
  FeeCalculationFailed(parking_fee.ParkingFeeError)
  MissingPaymentId
  PaymentCreationFailed(payment.PaymentError)
  PaymentSettlementFailed(payment.PaymentError)
}

pub fn error_message(error: ExitParkingError) -> String {
  case error {
    SessionNotFound -> "Parking session was not found"

    SessionClosingFailed(session_error) ->
      parking_session.error_message(session_error)

    FeeCalculationFailed(fee_error) -> parking_fee.error_message(fee_error)

    MissingPaymentId ->
      "Payment ID is required when parking fee is greater than zero"

    PaymentCreationFailed(payment_error) -> payment.error_message(payment_error)

    PaymentSettlementFailed(payment_error) ->
      payment.error_message(payment_error)
  }
}

// MARK: Exit Parking Request

pub type ExitParkingRequest {
  ExitParkingRequest(
    session_id: parking_session.ParkingSessionId,
    payment_id: option.Option(payment.PaymentId),
    exit_time: Int,
    hourly_rate: Int,
    payment_method: payment.PaymentMethod,
  )
}

// MARK: Exit Parking Result

pub type ExitParkingResult {
  ExitParkingResult(
    session: parking_session.ParkingSession,
    payment: option.Option(payment.Payment),
    duration_seconds: Int,
    fee: Int,
  )
}

// MARK: Exit Parking

pub fn exit_parking(
  session: parking_session.ParkingSession,
  request: ExitParkingRequest,
) -> Result(ExitParkingResult, ExitParkingError) {
  let ExitParkingRequest(
    session_id: _,
    payment_id: payment_id,
    exit_time: exit_time,
    hourly_rate: hourly_rate,
    payment_method: payment_method,
  ) = request

  // Validate the session state and exit time first.
  use closed_session <- result.try(
    parking_session.close_parking_session(session, exit_time)
    |> result.map_error(SessionClosingFailed),
  )

  // Calculate the actual parking duration.
  let entry_time = parking_session.get_parking_session_entry_time(session)
  let duration_seconds = exit_time - entry_time

  // Calculate the parking fee.
  use fee <- result.try(
    parking_fee.calculate_parking_fee(duration_seconds, hourly_rate)
    |> result.map_error(FeeCalculationFailed),
  )

  // No payment is necessary when the fee is zero.
  case fee {
    0 ->
      Ok(ExitParkingResult(
        session: closed_session,
        payment: option.None,
        duration_seconds: duration_seconds,
        fee: fee,
      ))

    _ -> {
      // A positive fee requires a payment ID.
      use payment_id <- result.try(require_payment_id(payment_id))

      // Create a pending payment.
      use pending_payment <- result.try(
        payment.new_payment(
          payment_id,
          parking_session.get_parking_session_id(session),
          fee,
          payment_method,
          exit_time,
        )
        |> result.map_error(PaymentCreationFailed),
      )

      // Simulate successful payment settlement.
      use settled_payment <- result.try(
        payment.succeed_payment(pending_payment)
        |> result.map_error(PaymentSettlementFailed),
      )

      Ok(ExitParkingResult(
        session: closed_session,
        payment: option.Some(settled_payment),
        duration_seconds: duration_seconds,
        fee: fee,
      ))
    }
  }
}

// MARK: Payment ID Validation

fn require_payment_id(
  payment_id: option.Option(payment.PaymentId),
) -> Result(payment.PaymentId, ExitParkingError) {
  case payment_id {
    option.Some(id) -> Ok(id)
    option.None -> Error(MissingPaymentId)
  }
}

import application/ports/parking_floor_repository
import application/ports/repository_error
import domain/floor
import domain/parking_slot
import domain/vehicle
import gleam/list
import gleam/result

pub type SeedParkingLayoutError {
  InvalidFloorId(floor.ParkingFloorError)
  InvalidSlotId(parking_slot.ParkingSlotError)
  FloorConfigurationFailed(floor.ParkingFloorError)
  RepositoryFailed(repository_error.RepositoryError)
}

pub fn seed(
  repository: parking_floor_repository.ParkingFloorRepository,
) -> Result(Nil, SeedParkingLayoutError) {
  let parking_floor_repository.ParkingFloorRepository(
    save_floor: save_floor,
    ..,
  ) = repository

  use first_floor <- result.try(
    build_floor("F1", [
      #("M-01", vehicle.Motorcycle),
      #("M-02", vehicle.Motorcycle),
      #("C-01", vehicle.Car),
      #("C-02", vehicle.Car),
      #("T-01", vehicle.Truck),
    ]),
  )

  use _ <- result.try(
    save_floor(first_floor)
    |> result.map_error(RepositoryFailed),
  )

  use second_floor <- result.try(
    build_floor("F2", [
      #("M-03", vehicle.Motorcycle),
      #("C-03", vehicle.Car),
      #("C-04", vehicle.Car),
    ]),
  )

  save_floor(second_floor)
  |> result.map_error(RepositoryFailed)
}

fn build_floor(
  floor_id_value: String,
  slot_definitions: List(#(String, vehicle.VehicleType)),
) -> Result(floor.ParkingFloor, SeedParkingLayoutError) {
  use floor_id <- result.try(
    floor.new_parking_floor_id(floor_id_value)
    |> result.map_error(InvalidFloorId),
  )

  let initial_floor = floor.new_parking_floor(floor_id)

  list.fold(slot_definitions, Ok(initial_floor), fn(acc, definition) {
    use current_floor <- result.try(acc)

    let #(slot_id_value, supported_vehicle_type) = definition

    use slot_id <- result.try(
      parking_slot.new_parking_slot_id(slot_id_value)
      |> result.map_error(InvalidSlotId),
    )

    let slot = parking_slot.new_parking_slot(slot_id, supported_vehicle_type)

    floor.add_parking_slot(current_floor, slot)
    |> result.map_error(FloorConfigurationFailed)
  })
}

import domain/parking_slot
import gleam/list
import gleam/string

// MARK: Errors

pub type ParkingFloorError {
  EmptyParkingFloorId
  ParkingSlotAlreadyOnFloor
}

pub fn error_message(error: ParkingFloorError) -> String {
  case error {
    EmptyParkingFloorId -> "Parking floor ID cannot be empty"

    ParkingSlotAlreadyOnFloor -> "Parking slot already exists on this floor"
  }
}

// MARK: Parking Floor ID

pub opaque type ParkingFloorId {
  ParkingFloorId(String)
}

pub fn new_parking_floor_id(
  value: String,
) -> Result(ParkingFloorId, ParkingFloorError) {
  let trimmed = string.trim(value)

  case trimmed {
    "" -> Error(EmptyParkingFloorId)
    _ -> Ok(ParkingFloorId(trimmed))
  }
}

pub fn get_parking_floor_id_value(id: ParkingFloorId) -> String {
  let ParkingFloorId(value) = id
  value
}

// MARK: Parking Floor

pub opaque type ParkingFloor {
  ParkingFloor(id: ParkingFloorId, slots: List(parking_slot.ParkingSlot))
}

pub fn new_parking_floor(id: ParkingFloorId) -> ParkingFloor {
  ParkingFloor(id: id, slots: [])
}

// MARK: Accessors

pub fn get_parking_floor_id(floor: ParkingFloor) -> ParkingFloorId {
  let ParkingFloor(id: id, ..) = floor
  id
}

pub fn get_parking_floor_slots(
  floor: ParkingFloor,
) -> List(parking_slot.ParkingSlot) {
  let ParkingFloor(slots: slots, ..) = floor
  slots
}

pub fn get_parking_floor_slot_count(floor: ParkingFloor) -> Int {
  list.length(get_parking_floor_slots(floor))
}

// MARK: Add Slot

pub fn add_parking_slot(
  floor: ParkingFloor,
  slot: parking_slot.ParkingSlot,
) -> Result(ParkingFloor, ParkingFloorError) {
  let slots = get_parking_floor_slots(floor)
  let slot_id = parking_slot.get_parking_slot_id(slot)

  let already_exists =
    list.any(slots, fn(existing_slot) {
      parking_slot.get_parking_slot_id(existing_slot) == slot_id
    })

  case already_exists {
    True -> Error(ParkingSlotAlreadyOnFloor)

    False -> Ok(ParkingFloor(..floor, slots: list.append(slots, [slot])))
  }
}

import domain/parking_session
import domain/parking_slot
import domain/vehicle
import gleam/list
import gleam/result

// MARK: Errors

pub type ParkingAllocationError {
  VehicleAlreadyParked
  ParkingSlotOccupied
  VehicleTypeNotSupported
  InvalidEntryTime
  InvalidParkingSession
}

pub type ParkingAllocationResult(a) =
  Result(a, ParkingAllocationError)

pub fn error_message(error: ParkingAllocationError) -> String {
  case error {
    VehicleAlreadyParked -> "Vehicle already has an active parking session"

    ParkingSlotOccupied -> "Parking slot is already occupied"

    VehicleTypeNotSupported ->
      "Vehicle type is not supported by this parking slot"

    InvalidEntryTime -> "Entry time cannot be negative"

    InvalidParkingSession -> "Could not create a valid parking session"
  }
}

// MARK: Allocation Request

pub type ParkingAllocationRequest {
  ParkingAllocationRequest(
    session_id: parking_session.ParkingSessionId,
    vehicle: vehicle.Vehicle,
    slot: parking_slot.ParkingSlot,
    entry_time: Int,
  )
}

// MARK: Allocation

pub fn allocate_parking_session(
  existing_sessions: List(parking_session.ParkingSession),
  request: ParkingAllocationRequest,
) -> ParkingAllocationResult(parking_session.ParkingSession) {
  let ParkingAllocationRequest(
    session_id: session_id,
    vehicle: candidate_vehicle,
    slot: slot,
    entry_time: entry_time,
  ) = request

  let vehicle_id = vehicle.get_vehicle_id(candidate_vehicle)
  let slot_id = parking_slot.get_parking_slot_id(slot)

  use _ <- result.try(validate_vehicle_type(candidate_vehicle, slot))

  use _ <- result.try(validate_vehicle_availability(
    existing_sessions,
    vehicle_id,
  ))

  use _ <- result.try(validate_slot_availability(existing_sessions, slot_id))

  construct_parking_session(session_id, vehicle_id, slot_id, entry_time)
}

// MARK: Validation

fn validate_vehicle_type(
  candidate_vehicle: vehicle.Vehicle,
  slot: parking_slot.ParkingSlot,
) -> ParkingAllocationResult(Nil) {
  case parking_slot.supports_vehicle(slot, candidate_vehicle) {
    True -> Ok(Nil)
    False -> Error(VehicleTypeNotSupported)
  }
}

fn validate_vehicle_availability(
  sessions: List(parking_session.ParkingSession),
  vehicle_id: vehicle.VehicleId,
) -> ParkingAllocationResult(Nil) {
  case vehicle_has_active_session(sessions, vehicle_id) {
    True -> Error(VehicleAlreadyParked)
    False -> Ok(Nil)
  }
}

fn validate_slot_availability(
  sessions: List(parking_session.ParkingSession),
  slot_id: parking_slot.ParkingSlotId,
) -> ParkingAllocationResult(Nil) {
  case slot_has_active_session(sessions, slot_id) {
    True -> Error(ParkingSlotOccupied)
    False -> Ok(Nil)
  }
}

// MARK: Active Session Queries

fn vehicle_has_active_session(
  sessions: List(parking_session.ParkingSession),
  vehicle_id: vehicle.VehicleId,
) -> Bool {
  list.any(sessions, fn(session) {
    case parking_session.get_parking_session_state(session) {
      parking_session.Active ->
        parking_session.get_parking_session_vehicle_id(session) == vehicle_id

      parking_session.Closed(_) -> False
    }
  })
}

fn slot_has_active_session(
  sessions: List(parking_session.ParkingSession),
  slot_id: parking_slot.ParkingSlotId,
) -> Bool {
  list.any(sessions, fn(session) {
    case parking_session.get_parking_session_state(session) {
      parking_session.Active ->
        parking_session.get_parking_session_slot_id(session) == slot_id

      parking_session.Closed(_) -> False
    }
  })
}

// MARK: Session Construction

fn construct_parking_session(
  session_id: parking_session.ParkingSessionId,
  vehicle_id: vehicle.VehicleId,
  slot_id: parking_slot.ParkingSlotId,
  entry_time: Int,
) -> ParkingAllocationResult(parking_session.ParkingSession) {
  case
    parking_session.new_parking_session(
      session_id,
      vehicle_id,
      slot_id,
      entry_time,
    )
  {
    Ok(session) -> Ok(session)

    Error(parking_session.InvalidEntryTime) -> Error(InvalidEntryTime)

    Error(_) -> Error(InvalidParkingSession)
  }
}

import gleam/int

// MARK: Errors

pub type ParkingFeeError {
  InvalidDuration
  InvalidHourlyRate
}

pub fn error_message(error: ParkingFeeError) -> String {
  case error {
    InvalidDuration -> "Parking duration cannot be negative"
    InvalidHourlyRate -> "Hourly rate must be greater than zero"
  }
}

// MARK: Constants

const seconds_per_hour = 3600

const rounding_offset = 1800

// MARK: Fee Calculation

/// Calculates a parking fee proportionally to the actual duration.
///
/// The result is rounded to the nearest whole rupiah.
/// A half-rupiah is rounded up.
///
/// Example:
/// - 10 minutes at Rp5,000/hour = Rp833
/// - 30 minutes at Rp5,000/hour = Rp2,500
pub fn calculate_parking_fee(
  duration_seconds: Int,
  hourly_rate: Int,
) -> Result(Int, ParkingFeeError) {
  case duration_seconds < 0 {
    True -> Error(InvalidDuration)
    False -> {
      case hourly_rate <= 0 {
        True -> Error(InvalidHourlyRate)
        False -> {
          let numerator = duration_seconds * hourly_rate + rounding_offset

          // The denominator is a non-zero constant, so division
          // cannot fail.
          let assert Ok(fee) = int.divide(numerator, seconds_per_hour)

          Ok(fee)
        }
      }
    }
  }
}

import application/ports/repository_error
import sqlight

/// Membuka koneksi ke database utama.
pub fn connect() -> Result(sqlight.Connection, repository_error.RepositoryError) {
  connect_to("priv/repo/parking_lot.db")
}

/// Membuka koneksi ke database berdasarkan path.
///
/// Foreign key diaktifkan untuk setiap koneksi.
pub fn connect_to(
  path: String,
) -> Result(sqlight.Connection, repository_error.RepositoryError) {
  case sqlight.open(path) {
    Ok(connection) -> {
      case sqlight.exec("PRAGMA foreign_keys = ON;", on: connection) {
        Ok(Nil) -> Ok(connection)

        Error(error) -> {
          let storage_error = to_repository_error(error)

          case sqlight.close(connection) {
            Ok(Nil) -> Error(storage_error)
            Error(_) -> Error(storage_error)
          }
        }
      }
    }

    Error(error) -> Error(to_repository_error(error))
  }
}

/// Mengubah error SQLight menjadi RepositoryError.
fn to_repository_error(
  error: sqlight.Error,
) -> repository_error.RepositoryError {
  case error {
    sqlight.SqlightError(_, message, _) ->
      repository_error.StorageError(message)
  }
}

// MARK: Enter Parking

pub fn enter(
  repository: SqliteEntryParkingTransaction,
  request: entry_parking.EntryParkingRequest,
) -> Result(
  entry_parking.EntryParkingResult,
  entry_parking_transaction.EntryParkingTransactionError,

  with_transaction(db, fn() { enter_in_transaction(db, request) })

// MARK: Transaction Lifecycle

fn with_transaction(
  db: sqlight.Connection,
  operation: fn() ->
    Result(a, entry_parking_transaction.EntryParkingTransactionError),
) -> Result(a, entry_parking_transaction.EntryParkingTransactionError) {
  use _ <- result.try(
    begin_transaction(db)
    |> result.map_error(entry_parking_transaction.RepositoryFailed),
  )

  finish_transaction(db, operation())
}

fn finish_transaction(
  db: sqlight.Connection,
  operation: Result(a, entry_parking_transaction.EntryParkingTransactionError),
) -> Result(a, entry_parking_transaction.EntryParkingTransactionError) {
  case operation {
    Ok(value) -> commit_or_rollback(db, value)
    Error(operation_error) -> rollback_after_failure(db, operation_error)
  }
}

fn rollback_after_failure(
  db: sqlight.Connection,
  operation_error: entry_parking_transaction.EntryParkingTransactionError,
) -> Result(a, entry_parking_transaction.EntryParkingTransactionError) {
  case rollback_transaction(db) {
    Ok(Nil) -> Error(operation_error)

    Error(rollback_error) ->
      Error(entry_parking_transaction.RollbackFailed(
        operation_error: operation_error,
        rollback_error: rollback_error,
      ))
  }
}

fn commit_or_rollback(
  db: sqlight.Connection,
  value: a,
) -> Result(a, entry_parking_transaction.EntryParkingTransactionError) {
  case commit_transaction(db) {
    Ok(Nil) -> Ok(value)

    Error(commit_error) ->
      rollback_after_failure(
        db,
        entry_parking_transaction.RepositoryFailed(commit_error),
      )
  }
}

// MARK: Entry Operation

fn enter_in_transaction(
  db: sqlight.Connection,
  request: entry_parking.EntryParkingRequest,
) -> Result(
  entry_parking.EntryParkingResult,
  entry_parking_transaction.EntryParkingTransactionError,
) {
  let entry_parking.EntryParkingRequest(vehicle: candidate_vehicle, ..) =
    request

  use _ <- result.try(
    persist_vehicle(db, candidate_vehicle)
    |> result.map_error(entry_parking_transaction.RepositoryFailed),
  )

  let floor_repository = sqlite_floor_repository.new(db)
  let session_repository = sqlite_session_repository.new(db)

  use floors <- result.try(
    sqlite_floor_repository.get_all_floors(floor_repository)
    |> result.map_error(entry_parking_transaction.RepositoryFailed),
  )

  use active_sessions <- result.try(
    sqlite_session_repository.get_active_sessions(session_repository)
    |> result.map_error(entry_parking_transaction.RepositoryFailed),
  )

  use entry_result <- result.try(
    entry_parking.enter_parking(floors, active_sessions, request)
    |> result.map_error(entry_parking_transaction.EntryFailed),
  )

  use _ <- result.try(
    persist_session(db, entry_result.session)
    |> result.map_error(entry_parking_transaction.RepositoryFailed),
  )

  use _ <- result.try(
    persist_ticket(db, entry_result.ticket)
    |> result.map_error(entry_parking_transaction.RepositoryFailed),
  )

  Ok(entry_result)
}

fn with_transaction(
  db: sqlight.Connection,
  operation: fn() ->
    Result(a, exit_parking_transaction.ExitParkingTransactionError),

  use _ <- result.try(
    begin_transaction(db)
    |> result.map_error(exit_parking_transaction.RepositoryFailed),
  )

fn finish_transaction(
  db: sqlight.Connection,
  operation: Result(a, exit_parking_transaction.ExitParkingTransactionError),

    Ok(value) -> commit_or_rollback(db, value)
    Error(operation_error) -> rollback_after_failure(db, operation_error)

    Error(rollback_error) ->
      Error(exit_parking_transaction.RollbackFailed(
        operation_error: operation_error,
        rollback_error: rollback_error,
      ))
  }
}

  floor_id: String,

  let slot_id =
    slot
    |> parking_slot.get_parking_slot_id
    |> parking_slot.get_parking_slot_id_value

    slot
    |> parking_slot.get_supported_vehicle_type
    |> vehicle_type_to_string

  let sql =

    <> escape_sql(floor_id)
    <> "', '"
    <> vehicle_type
    <> "') ON CONFLICT(id) DO NOTHING;"

fn verify_slot_configuration(
  db: sqlight.Connection,
  expected_floor_id: String,
  slot_id: String,
  expected_vehicle_type: String,
) -> Result(Nil, repository_error.RepositoryError) {
  let sql =
    "SELECT floor_id, supported_vehicle_type "
    <> "FROM parking_slots WHERE id = '"
    <> escape_sql(slot_id)
    <> "';"

  use rows <- result.try(
    query(sqlight.query(
      sql,
      on: db,
      with: [],
      expecting: slot_configuration_decoder(),
    )),
  )

  case rows {
    [#(actual_floor_id, actual_vehicle_type)] ->
      case
        actual_floor_id == expected_floor_id
        && actual_vehicle_type == expected_vehicle_type
      {
        True -> Ok(Nil)

        False ->
          Error(repository_error.StorageError(
            "Parking slot configuration conflict: " <> slot_id,
          ))
      }

    _ ->
      Error(repository_error.StorageError(
        "Unable to verify parking slot: " <> slot_id,
      ))
  }
}

fn attach_slots(

  slots: List(#(String, String, String)),

  list.fold(slots, Ok(floors), fn(acc, slot_data) {

fn floor_id_value(current_floor: floor.ParkingFloor) -> String {
  current_floor
  |> floor.get_parking_floor_id
  |> floor.get_parking_floor_id_value
}

fn add_slot_to_floors(
  floors: List(floor.ParkingFloor),
  floor_id: String,
  slot: parking_slot.ParkingSlot,
) -> Result(List(floor.ParkingFloor), repository_error.RepositoryError) {
  let floor_exists =
    list.any(floors, fn(current_floor) {
      floor_id_value(current_floor) == floor_id
    })

  case floor_exists {
    False ->
      Error(repository_error.StorageError(
        "Parking slot references an unknown floor: " <> floor_id,
      ))

    True ->
      floors
      |> list.map(fn(current_floor) {
        add_slot_to_matching_floor(current_floor, floor_id, slot)
      })
      |> result.all
  }
}

fn add_slot_to_matching_floor(
  current_floor: floor.ParkingFloor,
  target_floor_id: String,
  slot: parking_slot.ParkingSlot,
) -> Result(floor.ParkingFloor, repository_error.RepositoryError) {
  case floor_id_value(current_floor) == target_floor_id {
    True ->
      floor.add_parking_slot(current_floor, slot)
      |> result.map_error(fn(error) {
        repository_error.StorageError(floor.error_message(error))
      })

    False -> Ok(current_floor)
  }
}

      Error(repository_error.StorageError(message))
  }

import gleam/dynamic/decode
import gleam/list
import gleam/result

import migrant
import sqlight

const required_tables = [
  "parking_floors",
  "parking_slots",
  "vehicles",
  "parking_sessions",
  "parking_tickets",
]

const initial_migration_name = "001_create_parking_schema"

/// Menjalankan migrasi database.
///
/// Database baru akan dimigrasikan menggunakan Migrant.
/// Database lama dengan legacy migration history akan diadopsi
/// ke Migrant sebelum menjalankan migrasi yang masih pending.
pub fn migrate(
  db: sqlight.Connection,
  migrations_dir: String,
) -> Result(Nil, String) {
  use tables <- result.try(load_table_names(db))

  let existing_domain_tables =
    list.filter(tables, fn(table) { list.contains(required_tables, table) })

  case list.length(existing_domain_tables) {
    0 -> migrate_new_database(db, tables, migrations_dir)

    5 -> migrate_existing_database(db, tables, migrations_dir)

    _ ->
      Error(
        "Schema database tidak lengkap. Migrasi dihentikan untuk mencegah perubahan schema yang tidak aman.",
      )
  }
}

fn migrate_new_database(
  db: sqlight.Connection,
  tables: List(String),
  migrations_dir: String,
) -> Result(Nil, String) {
  case list.contains(tables, "schema_migrations") {
    True ->
      Error(
        "Database memiliki schema_migrations tetapi tidak memiliki schema domain. Pemeriksaan manual diperlukan.",
      )

    False ->
      case list.contains(tables, "__migrations") {
        True ->
          Error(
            "Database memiliki __migrations tetapi tidak memiliki schema domain. Pemeriksaan manual diperlukan.",
          )

        False -> run_pending_migrations(db, migrations_dir)
      }
  }
}

fn migrate_existing_database(
  db: sqlight.Connection,
  tables: List(String),
  migrations_dir: String,
) -> Result(Nil, String) {
  case list.contains(tables, "__migrations") {
    True -> {
      use applied <- result.try(
        migrant_migration_is_applied(db)
        |> result.map_error(fn(_) {
          "Gagal memeriksa migration history Migrant."
        }),
      )

      case applied {
        True -> run_pending_migrations(db, migrations_dir)

        False ->
          Error(
            "Schema domain sudah ada tetapi migration history Migrant tidak mencatat migrasi awal.",
          )
      }
    }

    False -> {
      use _ <- result.try(adopt_legacy_database(db, tables))

      run_pending_migrations(db, migrations_dir)
    }
  }
}

fn adopt_legacy_database(
  db: sqlight.Connection,
  tables: List(String),
) -> Result(Nil, String) {
  case list.contains(tables, "schema_migrations") {
    False ->
      Error(
        "Schema domain sudah ada tetapi legacy migration history tidak ditemukan.",
      )

    True -> {
      use legacy_applied <- result.try(
        legacy_migration_is_applied(db)
        |> result.map_error(fn(_) {
          "Gagal memeriksa legacy migration history."
        }),
      )

      case legacy_applied {
        False ->
          Error(
            "Schema domain sudah ada tetapi migrasi awal tidak tercatat pada legacy migration history.",
          )

        True -> record_migrant_migration(db)
      }
    }
  }
}

/// Menjalankan seluruh migrasi yang belum tercatat oleh Migrant.
fn run_pending_migrations(
  db: sqlight.Connection,
  migrations_dir: String,
) -> Result(Nil, String) {
  migrant.migrate(db, migrations_dir)
  |> result.map_error(fn(_) {
    "Gagal menjalankan migrasi database menggunakan Migrant."
  })
}

fn load_table_names(db: sqlight.Connection) -> Result(List(String), String) {
  sqlight.query(
    "
    SELECT name
    FROM sqlite_master
    WHERE type = 'table'
      AND name NOT LIKE 'sqlite_%';
    ",
    on: db,
    with: [],
    expecting: decode.field(0, decode.string, fn(name) { decode.success(name) }),
  )
  |> result.map_error(fn(_) { "Gagal membaca daftar tabel database." })
}

fn legacy_migration_is_applied(
  db: sqlight.Connection,
) -> Result(Bool, sqlight.Error) {
  use versions <- result.try(sqlight.query(
    "
    SELECT name
    FROM schema_migrations
    WHERE version = 1;
    ",
    on: db,
    with: [],
    expecting: decode.field(0, decode.string, fn(name) { decode.success(name) }),
  ))

  Ok(list.contains(versions, "001_create_parking_schema.sql"))
}

fn migrant_migration_is_applied(
  db: sqlight.Connection,
) -> Result(Bool, sqlight.Error) {
  use names <- result.try(sqlight.query(
    "
    SELECT name
    FROM __migrations
    WHERE name = ?;
    ",
    on: db,
    with: [sqlight.text(initial_migration_name)],
    expecting: decode.field(0, decode.string, fn(name) { decode.success(name) }),
  ))

  Ok(list.contains(names, initial_migration_name))
}

fn record_migrant_migration(db: sqlight.Connection) -> Result(Nil, String) {
  case sqlight.exec("BEGIN IMMEDIATE;", on: db) {
    Error(_) -> Error("Gagal memulai transaksi adopsi migration.")

    Ok(Nil) -> {
      let outcome = record_migrant_migration_in_transaction(db)

      case outcome {
        Ok(Nil) ->
          case sqlight.exec("COMMIT;", on: db) {
            Ok(Nil) -> Ok(Nil)

            Error(_) -> {
              rollback_migration_adoption(db)
              Error("Gagal melakukan commit adopsi migration.")
            }
          }

        Error(message) -> {
          rollback_migration_adoption(db)
          Error(message)
        }
      }
    }
  }
}

fn record_migrant_migration_in_transaction(
  db: sqlight.Connection,
) -> Result(Nil, String) {
  use _ <- result.try(
    sqlight.exec(
      "
      CREATE TABLE IF NOT EXISTS __migrations (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      );
      ",
      on: db,
    )
    |> result.map_error(fn(_) {
      "Gagal membuat tabel migration history Migrant."
    }),
  )

  use _ <- result.try(
    sqlight.exec(
      "
      INSERT INTO __migrations (name)
      SELECT '001_create_parking_schema'
      WHERE NOT EXISTS (
        SELECT 1
        FROM __migrations
        WHERE name = '001_create_parking_schema'
      );
      ",
      on: db,
    )
    |> result.map_error(fn(_) {
      "Gagal mengadopsi legacy migration ke Migrant."
    }),
  )

  Ok(Nil)
}

fn rollback_migration_adoption(db: sqlight.Connection) {
  let _ = sqlight.exec("ROLLBACK;", on: db)

  Nil
}

pub fn get_payment(
  repository: SqlitePaymentRepository,
  id: payment.PaymentId,

  let SqlitePaymentRepository(db: db) = repository

  use payment_data <- result.try(load_payment(db, id))

    [data] -> {
      use value <- result.try(build_payment_from_data(data))
      Ok(option.Some(value))

    [_, _, ..] ->
      Error(repository_error.StorageError(
        "Multiple payments found for one payment ID",

pub fn save_payment(
  repository: SqlitePaymentRepository,
  value: payment.Payment,
) -> Result(Nil, repository_error.RepositoryError) {
  let SqlitePaymentRepository(db: db) = repository

  let payment_id = payment.get_payment_id_value(payment.get_payment_id(value))

  let session_id =
    parking_session.get_parking_session_id_value(payment.get_payment_session_id(
      value,
    ))

  let amount = payment.get_payment_amount(value)
  let method = payment.get_payment_method(value)
  let status = payment.get_payment_status(value)
  let created_at = payment.get_payment_created_at(value)

  use method_value <- result.try(payment_method_to_string(method))
  use status_value <- result.try(payment_status_to_string(status))

  use saved_ids <- result.try(
    query(sqlight.query(
      "
    INSERT INTO payments (
      id,
      session_id,
      amount,
      method,
      status,
      created_at
    )
    VALUES (?, ?, ?, ?, ?, ?)
    ON CONFLICT(id) DO UPDATE SET
      session_id = excluded.session_id,
      amount = excluded.amount,
      method = excluded.method,
      status = excluded.status,
      created_at = excluded.created_at
    RETURNING id;
    ",
      on: db,
      with: [
        sqlight.text(payment_id),
        sqlight.text(session_id),
        sqlight.int(amount),
        sqlight.text(method_value),
        sqlight.text(status_value),
        sqlight.int(created_at),
      ],
      expecting: decode.field(0, decode.string, fn(id) { decode.success(id) }),
    )),
  )

  case saved_ids {
    [saved_id] if saved_id == payment_id -> Ok(Nil)

    _ -> Error(repository_error.StorageError("Failed to persist payment"))
  }
}

fn payment_decoder() {
  decode.at([0], decode.string)
  |> decode.then(fn(payment_id) {
    decode.at([1], decode.string)
    |> decode.then(fn(session_id) {
      decode.at([2], decode.int)
      |> decode.then(fn(amount) {
        decode.at([3], decode.string)
        |> decode.then(fn(method) {
          decode.at([4], decode.string)
          |> decode.then(fn(status) {
            decode.at([5], decode.int)
            |> decode.map(fn(created_at) {
              #(payment_id, session_id, amount, method, status, created_at)
            })
          })
        })
      })
    })
  })

fn build_payment_from_data(
  data: #(String, String, Int, String, String, Int),
) -> Result(payment.Payment, repository_error.RepositoryError) {
  let #(payment_id, session_id, amount, method, status, created_at) = data

  use id <- result.try(build_payment_id(payment_id))
  use session_id <- result.try(build_session_id(session_id))
  use method <- result.try(payment_method_from_string(method))

  use value <- result.try(
    case payment.new_payment(id, session_id, amount, method, created_at) {
      Ok(value) -> Ok(value)

      Error(error) ->
        Error(repository_error.StorageError(payment.error_message(error)))
    },
  )

  apply_payment_status(value, status)
}

  case payment.new_payment_id(value) {

  }
}

}

fn apply_payment_status(
  value: payment.Payment,
  status: String,
) -> Result(payment.Payment, repository_error.RepositoryError) {
  case status {
    "Pending" -> Ok(value)

    "Succeeded" ->
      case payment.succeed_payment(value) {
        Ok(updated) -> Ok(updated)

        Error(error) ->
          Error(repository_error.StorageError(payment.error_message(error)))
      }

    "Failed" ->
      case payment.fail_payment(value) {
        Ok(updated) -> Ok(updated)

        Error(error) ->
          Error(repository_error.StorageError(payment.error_message(error)))
      }

    "Cancelled" ->
      case payment.cancel_payment(value) {
        Ok(updated) -> Ok(updated)

        Error(error) ->
          Error(repository_error.StorageError(payment.error_message(error)))
      }

    _ ->
      Error(repository_error.StorageError("Invalid payment status in database"))
  }
}

  }

  slots: List(parking_slot.ParkingSlot),
) -> floor.ParkingFloor {
  let assert Ok(floor_id) = floor.new_parking_floor_id(id)
  let parking_floor = floor.new_parking_floor(floor_id)

fn add_slots(
  parking_floor: floor.ParkingFloor,
  slots: List(parking_slot.ParkingSlot),
) -> floor.ParkingFloor {
  case slots {

      let assert Ok(updated_floor) = floor.add_parking_slot(parking_floor, slot)

      add_slots(updated_floor, remaining_slots)
    }
  }
}

// MARK: Atomic Rollback

pub fn failed_session_update_rolls_back_payment_test() {
  let db = setup_database()

  // Payment persistence happens before session persistence.
  // This trigger forces the second operation to fail.
  let _ =
    execute(
      db,
      "
    CREATE TRIGGER reject_session_exit
    BEFORE UPDATE ON parking_sessions
    BEGIN
      SELECT RAISE(ABORT, 'Simulated session update failure');
    END;
    ",
    )

  let transaction = sqlite_exit_parking_transaction.new(db)
  let request = make_request(Some(make_payment_id()), 4600, 5000)

  let result = sqlite_exit_parking_transaction.exit(transaction, request)

  case result {
    Error(exit_parking_transaction.RepositoryFailed(repository_error.StorageError(
      _,
    ))) -> Nil

    Error(_) -> should.fail()
    Ok(_) -> should.fail()
  }

  // The payment INSERT must be rolled back as well.
  count_rows(db, "payments")
  |> should.equal(0)

  // The original session must remain active.
  get_session_state(db)
  |> should.equal("Active")

  close_database(db)
}

fn table_exists(db: sqlight.Connection, table_name: String) -> Bool {
  let assert Ok(rows) =
    sqlight.query(
      "
      SELECT name
      FROM sqlite_master
      WHERE type = 'table'
        AND name = ?;
      ",
      on: db,
      with: [sqlight.text(table_name)],
      expecting: decode.at([0], decode.string),
    )

  list.contains(rows, table_name)
}

fn migration_is_recorded(db: sqlight.Connection) -> Bool {
  let assert Ok(names) =
    sqlight.query(
      "
      SELECT name
      FROM __migrations
      WHERE name = ?;
      ",
      on: db,
      with: [sqlight.text(initial_migration_name)],
      expecting: decode.at([0], decode.string),
    )

  list.contains(names, initial_migration_name)
}
